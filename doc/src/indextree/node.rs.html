1//! Tree node and its associated data.
2//!
3//! Each [`Node`] stores its data along with links to its parent, children,
4//! and siblings. Nodes are not accessed directly; instead, use
5//! [`NodeId`](crate::NodeId) to index into an [`Arena`](crate::Arena).

18#[derive(PartialEq, Eq, Clone, Debug)]
19#[cfg_attr(feature = "serde", derive(Deserialize, Serialize))]
20pub(crate) enum NodeData<T> {
21    /// The actual data store
22    Data(T),
23    /// The next free node position.
24    NextFree(Option<usize>),
25}

27#[derive(PartialEq, Eq, Clone, Debug)]
28#[cfg_attr(feature = "serde", derive(Deserialize, Serialize))]
29/// A node within a particular `Arena`.
30pub struct Node<T> {
31    // Keep these private (with read-only accessors) so that we can keep them
32    // consistent. E.g. the parent of a node’s child is that node.
33    pub(crate) parent: Option<NodeId>,
34    pub(crate) previous_sibling: Option<NodeId>,
35    pub(crate) next_sibling: Option<NodeId>,
36    pub(crate) first_child: Option<NodeId>,
37    pub(crate) last_child: Option<NodeId>,
38    pub(crate) stamp: NodeStamp,
39    /// The actual data which will be stored within the tree.
40    pub(crate) data: NodeData<T>,
41}
42
43impl<T> Node<T> {
44    /// Returns a reference to the node data.
45    ///
46    /// # Panics
47    ///
48    /// Panics if the node has been removed.
49    ///
50    /// # Examples
51    ///
52    /// ```
53    /// # use indextree::Arena;
54    /// let mut arena = Arena::new();
55    /// let id = arena.new_node(42);
56    /// assert_eq!(*arena[id].get(), 42);
57    /// ```
58    #[inline]
59    pub fn get(&self) -> &T {
60        if let NodeData::Data(ref data) = self.data {
61            data
62        } else {
63            panic!("attempted to access a freed node")
64        }
65    }
66
67    /// Returns a reference to the node data, or `None` if the node has
68    /// been removed.
69    ///
70    /// # Examples
71    ///
72    /// ```
73    /// # use indextree::Arena;
74    /// let mut arena = Arena::new();
75    /// let id = arena.new_node(42);
76    /// assert_eq!(arena[id].try_get(), Some(&42));
77    /// ```
78    #[inline]
79    pub fn try_get(&self) -> Option<&T> {
80        if let NodeData::Data(ref data) = self.data {
81            Some(data)
82        } else {
83            None
84        }
85    }
86
87    /// Returns a mutable reference to the node data.
88    ///
89    /// # Panics
90    ///
91    /// Panics if the node has been removed.
92    ///
93    /// # Examples
94    ///
95    /// ```
96    /// # use indextree::Arena;
97    /// let mut arena = Arena::new();
98    /// let id = arena.new_node(42);
99    /// *arena[id].get_mut() = 99;
100    /// assert_eq!(*arena[id].get(), 99);
101    /// ```
102    #[inline]
103    pub fn get_mut(&mut self) -> &mut T {
104        if let NodeData::Data(ref mut data) = self.data {
105            data
106        } else {
107            panic!("attempted to access a freed node")
108        }
109    }
110
111    /// Returns a mutable reference to the node data, or `None` if the
112    /// node has been removed.
113    ///
114    /// # Examples
115    ///
116    /// ```
117    /// # use indextree::Arena;
118    /// let mut arena = Arena::new();
119    /// let id = arena.new_node(42);
120    /// if let Some(data) = arena[id].try_get_mut() {
121    ///     *data = 99;
122    /// }
123    /// assert_eq!(*arena[id].get(), 99);
124    /// ```
125    #[inline]
126    pub fn try_get_mut(&mut self) -> Option<&mut T> {
127        if let NodeData::Data(ref mut data) = self.data {
128            Some(data)
129        } else {
130            None
131        }
132    }
133
134    /// Consumes the node and returns the contained data, or `None` if
135    /// the node has been removed.
136    ///
137    /// # Examples
138    ///
139    /// ```
140    /// # use indextree::Arena;
141    /// let mut arena = Arena::new();
142    /// arena.new_node(42);
143    /// arena.new_node(99);
144    ///
145    /// let values: Vec<_> = arena
146    ///     .into_iter()
147    ///     .filter_map(|n| n.into_data())
148    ///     .collect();
149    /// assert_eq!(values, vec![42, 99]);
150    /// ```
151    pub fn into_data(self) -> Option<T> {
152        match self.data {
153            NodeData::Data(data) => Some(data),
154            NodeData::NextFree(_) => None,
155        }
156    }
157
158    /// Creates a new `Node` with the default state and the given data.
159    #[inline]
160    pub(crate) fn new(data: T) -> Self {
161        Self {
162            parent: None,
163            previous_sibling: None,
164            next_sibling: None,
165            first_child: None,
166            last_child: None,
167            stamp: NodeStamp::default(),
168            data: NodeData::Data(data),
169        }
170    }
171
172    /// Convert a removed `Node` to normal with default state and given data.
173    #[inline]
174    pub(crate) fn reuse(&mut self, data: T) {
175        debug_assert!(matches!(self.data, NodeData::NextFree(_)));
176        debug_assert!(self.stamp.is_removed());
177        self.stamp.reuse();
178        self.parent = None;
179        self.previous_sibling = None;
180        self.next_sibling = None;
181        self.first_child = None;
182        self.last_child = None;
183        self.data = NodeData::Data(data);
184    }

186    /// Returns the ID of the parent node, unless this node is the root of the
187    /// tree.
188    ///
189    /// # Examples
190    ///
191    /// ```
192    /// # use indextree::Arena;
193    /// # let mut arena = Arena::new();
194    /// # let n1 = arena.new_node("1");
195    /// # let n1_1 = arena.new_node("1_1");
196    /// # let n1_2 = arena.new_node("1_2");
197    /// # n1.append(n1_2, &mut arena);
198    /// # let n1_3 = arena.new_node("1_3");
199    /// # n1.append(n1_3, &mut arena);
200    /// # n1.append(n1_1, &mut arena);
201    /// // arena
202    /// // `-- 1
203    /// //     |-- 1_1
204    /// //     |-- 1_2
205    /// //     `-- 1_3
206    /// assert_eq!(arena[n1].parent(), None);
207    /// assert_eq!(arena[n1_1].parent(), Some(n1));
208    /// assert_eq!(arena[n1_2].parent(), Some(n1));
209    /// assert_eq!(arena[n1_3].parent(), Some(n1));
210    /// ```
211    #[inline]
212    pub fn parent(&self) -> Option<NodeId> {
213        self.parent
214    }

274    /// Returns the ID of the previous sibling of this node, unless it is a
275    /// first child.
276    ///
277    /// # Examples
278    ///
279    /// ```
280    /// # use indextree::Arena;
281    /// # let mut arena = Arena::new();
282    /// # let n1 = arena.new_node("1");
283    /// # let n1_1 = arena.new_node("1_1");
284    /// # n1.append(n1_1, &mut arena);
285    /// # let n1_2 = arena.new_node("1_2");
286    /// # n1.append(n1_2, &mut arena);
287    /// # let n1_3 = arena.new_node("1_3");
288    /// # n1.append(n1_3, &mut arena);
289    /// // arena
290    /// // `-- 1
291    /// //     |-- 1_1
292    /// //     |-- 1_2
293    /// //     `-- 1_3
294    /// assert_eq!(arena[n1].previous_sibling(), None);
295    /// assert_eq!(arena[n1_1].previous_sibling(), None);
296    /// assert_eq!(arena[n1_2].previous_sibling(), Some(n1_1));
297    /// assert_eq!(arena[n1_3].previous_sibling(), Some(n1_2));
298    /// ```
299    ///
300    /// Note that newly created nodes are independent toplevel nodes, and they
301    /// are not siblings by default.
302    ///

304    /// # use indextree::Arena;

316    /// n1.insert_after(n2, &mut arena);
317    /// // arena

319    /// //     |-- 1
320    /// //     `-- 2

419    pub fn is_removed(&self) -> bool {
420        self.stamp.is_removed()
421    }

423    /// Checks if the node is detached.
424    pub(crate) fn is_detached(&self) -> bool {
425        self.parent.is_none() && self.previous_sibling.is_none() && self.next_sibling.is_none()
426    }

429impl<T: hash::Hash> hash::Hash for Node<T> {
430    fn hash<H: hash::Hasher>(&self, state: &mut H) {
431        self.parent.hash(state);
432        self.previous_sibling.hash(state);
433        self.next_sibling.hash(state);
434        self.first_child.hash(state);
435        self.last_child.hash(state);
436        self.stamp.hash(state);
437        match self.data {
438            NodeData::Data(ref data) => {
439                0u8.hash(state);
440                data.hash(state);
441            }
442            NodeData::NextFree(ref next) => {
443                1u8.hash(state);
444                next.hash(state);
445            }
446        }
447    }
448}

450impl<T> fmt::Display for Node<T> {
451    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
452        if let Some(parent) = self.parent {
453            write!(f, "parent: {parent}; ")?;
454        } else {
455            write!(f, "no parent; ")?;
456        }
457        if let Some(previous_sibling) = self.previous_sibling {
458            write!(f, "previous sibling: {previous_sibling}; ")?;
459        } else {
460            write!(f, "no previous sibling; ")?;
461        }
462        if let Some(next_sibling) = self.next_sibling {
463            write!(f, "next sibling: {next_sibling}; ")?;
464        } else {
465            write!(f, "no next sibling; ")?;
466        }
467        if let Some(first_child) = self.first_child {
468            write!(f, "first child: {first_child}; ")?;
469        } else {
470            write!(f, "no first child; ")?;
471        }
472        if let Some(last_child) = self.last_child {
473            write!(f, "last child: {last_child}; ")?;
474        } else {
475            write!(f, "no last child; ")?;
476        }
477        Ok(())
478    }
