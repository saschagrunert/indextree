1//! Arena structure and node storage.
2//!
3//! The [`Arena`] is the central owner of all tree nodes. Nodes are stored
4//! contiguously in a single `Vec` and referenced by [`NodeId`](crate::NodeId).
5//! Removed nodes are recycled through an internal free list.

109    /// Retrieves the `NodeId` corresponding to a `Node` in the `Arena`.
110    ///
111    /// # Examples
112    ///
113    /// ```
114    /// # use indextree::Arena;
115    /// let mut arena = Arena::new();
116    /// let foo = arena.new_node("foo");
117    /// let node = arena.get(foo).unwrap();
118    ///
119    /// let node_id = arena.get_node_id(node).unwrap();
120    /// assert_eq!(*arena[node_id].get(), "foo");
121    /// ```
122    pub fn get_node_id(&self, node: &Node<T>) -> Option<NodeId> {
123        let nodes_range = self.nodes.as_ptr_range();
124        let p = node as *const Node<T>;
125
126        if !nodes_range.contains(&p) {
127            return None;
128        }
129
130        let node_index = (p as usize - nodes_range.start as usize) / mem::size_of::<Node<T>>();
131        let node_id = NonZeroUsize::new(node_index.wrapping_add(1))?;
132
133        Some(NodeId::from_non_zero_usize(
134            node_id,
135            self.nodes[node_index].stamp,
136        ))
137    }

139    /// Retrieves the `NodeId` corresponding to the `Node` at `index` in the `Arena`, if it exists.
140    ///
141    /// Note: We use 1 based indexing, so the first element is at `1` and not `0`.
142    ///
143    /// # Examples
144    ///
145    /// ```
146    /// # use indextree::Arena;
147    /// # use std::num::NonZeroUsize;
148    /// let mut arena = Arena::new();
149    /// let foo = arena.new_node("foo");
150    /// let node = arena.get(foo).unwrap();
151    /// let index: NonZeroUsize = foo.into();
152    ///
153    /// let new_foo = arena.get_node_id_at(index).unwrap();
154    /// assert_eq!(foo, new_foo);
155    ///
156    /// foo.remove(&mut arena);
157    /// let new_foo = arena.get_node_id_at(index);
158    /// assert!(new_foo.is_none(), "must be none if the node at the index doesn't exist");
159    /// ```
160    pub fn get_node_id_at(&self, index: NonZeroUsize) -> Option<NodeId> {
161        let index0 = index.get() - 1; // we use 1 based indexing.
162        self.nodes
163            .get(index0)
164            .filter(|n| !n.is_removed())
165            .map(|node| NodeId::from_non_zero_usize(index, node.stamp))
166    }

168    /// Creates a new node from its associated data.
169    ///
170    /// Freed slots are reused when available. If a slot's internal stamp has
171    /// been exhausted (after ~32K remove/reuse cycles), it is skipped and a
172    /// fresh slot is appended instead.
173    ///
174    /// # Panics
175    ///
176    /// Panics if the arena already has `usize::max_value()` nodes.
177    ///
178    /// # Examples
179    ///
180    /// ```
181    /// # use indextree::Arena;
182    /// let mut arena = Arena::new();
183    /// let foo = arena.new_node("foo");
184    ///
185    /// assert_eq!(*arena[foo].get(), "foo");
186    /// ```
187    pub fn new_node(&mut self, data: T) -> NodeId {
188        let (index, stamp) = if let Some(index) = self.pop_front_free_node() {
189            let node = &mut self.nodes[index];
190            node.reuse(data);
191            (index, node.stamp)
192        } else {
193            let index = self.nodes.len();
194            let node = Node::new(data);
195            let stamp = node.stamp;
196            self.nodes.push(node);
197            (index, stamp)
198        };
199        let next_index1 =
200            NonZeroUsize::new(index.wrapping_add(1)).expect("Too many nodes in the arena");
201        NodeId::from_non_zero_usize(next_index1, stamp)
202    }

278    /// Returns a reference to the node with the given id if in the arena.
279    ///
280    /// Returns `None` if the index is out of bounds or the node's stamp
281    /// does not match (i.e. the slot was removed and possibly reused).
282    ///
283    /// # Examples
284    ///
285    /// ```
286    /// # use indextree::{Arena, NodeId};
287    /// let mut arena = Arena::new();
288    /// let foo = arena.new_node("foo");
289    /// assert_eq!(arena.get(foo).map(|node| *node.get()), Some("foo"));
290    /// ```
291    ///
292    /// Stale `NodeId`s from removed nodes return `None`:
293    ///
294    /// ```
295    /// # use indextree::Arena;
296    /// let mut arena = Arena::new();
297    /// let foo = arena.new_node("foo");
298    /// foo.remove(&mut arena);
299    /// assert!(arena.get(foo).is_none());
300    /// ```
301    #[inline]
302    pub fn get(&self, id: NodeId) -> Option<&Node<T>> {
303        self.nodes
304            .get(id.index0())
305            .filter(|node| node.stamp == id.stamp())
306    }

417    /// Returns an iterator of [`NodeId`]s of all non-removed nodes in
418    /// the arena in storage-order.
419    ///
420    /// Unlike [`iter()`], this skips removed nodes and yields `NodeId`s
421    /// instead of `&Node<T>`.
422    ///
423    /// # Examples
424    ///
425    /// ```
426    /// # use indextree::Arena;
427    /// let mut arena = Arena::new();
428    /// let foo = arena.new_node("foo");
429    /// let bar = arena.new_node("bar");
430    /// let baz = arena.new_node("baz");
431    /// bar.remove(&mut arena);
432    ///
433    /// let ids: Vec<_> = arena.iter_node_ids().collect();
434    /// assert_eq!(ids, vec![foo, baz]);
435    /// ```
436    ///
437    /// [`iter()`]: Arena::iter
438    pub fn iter_node_ids(&self) -> impl Iterator<Item = NodeId> + '_ {
439        self.nodes.iter().enumerate().filter_map(|(i, node)| {
440            if node.is_removed() {
441                return None;
442            }
443            let index1 = NonZeroUsize::new(i.wrapping_add(1))?;
444            Some(NodeId::from_non_zero_usize(index1, node.stamp))
445        })
446    }

496    /// Creates a new arena by applying a function to the data of every
497    /// live node, preserving the tree structure.
498    ///
499    /// Removed nodes remain as removed slots in the new arena to keep
500    /// node indices consistent.
501    ///
502    /// # Examples
503    ///
504    /// ```
505    /// # use indextree::Arena;
506    /// let mut arena = Arena::new();
507    /// let root = arena.new_node(1);
508    /// let child = root.append_value(2, &mut arena);
509    ///
510    /// let mapped: Arena<String> = arena.map(|x| x.to_string());
511    /// assert_eq!(mapped.get_data(root), Some(&"1".to_string()));
512    /// assert_eq!(mapped.get_data(child), Some(&"2".to_string()));
513    /// assert_eq!(mapped[child].parent(), Some(root));
514    /// ```
515    pub fn map<U>(&self, mut f: impl FnMut(&T) -> U) -> Arena<U> {
516        let nodes = self
517            .nodes
518            .iter()
519            .map(|node| Node {
520                parent: node.parent,
521                previous_sibling: node.previous_sibling,
522                next_sibling: node.next_sibling,
523                first_child: node.first_child,
524                last_child: node.last_child,
525                stamp: node.stamp,
526                data: match &node.data {
527                    NodeData::Data(data) => NodeData::Data(f(data)),
528                    NodeData::NextFree(next) => NodeData::NextFree(*next),
529                },
530            })
531            .collect();
532        Arena {
533            nodes,
534            first_free_slot: self.first_free_slot,
535            last_free_slot: self.last_free_slot,
536        }
537    }

614    /// Validates the internal consistency of the arena's tree structure.
615    ///
616    /// Returns `true` if the arena passes all of the following checks:
617    ///
618    /// - Every live node contains actual data (not a free-list entry).
619    /// - Parent, previous/next sibling, and first/last child pointers
620    ///   all refer to valid, non-removed nodes with matching stamps.
621    /// - `first_child` and `last_child` are both set or both unset.
622    /// - Sibling back-pointers are reciprocal (prev's next == self,
623    ///   next's prev == self), and siblings share the same parent.
624    /// - Every child in a parent's child chain points back to that
625    ///   parent, and the chain ends at `last_child`.
626    /// - Every live node is reachable exactly once from a root sibling
627    ///   chain, which rules out cycles in parent and sibling links.
628    /// - The free list is well-formed: consistent first/last pointers,
629    ///   all entries are removed nodes with `NextFree` data, and no
630    ///   cycles.
631    ///
632    /// This is primarily useful after deserialization to detect
633    /// corrupted data.
634    ///
635    /// # Examples
636    ///
637    /// ```
638    /// # use indextree::Arena;
639    /// let mut arena = Arena::new();
640    /// let root = arena.new_node(1);
641    /// root.append_value(2, &mut arena);
642    /// assert!(arena.validate());
643    /// ```
644    #[must_use]
645    pub fn validate(&self) -> bool {
646        let len = self.nodes.len();
647
648        let is_valid = |id: NodeId| -> bool {
649            let idx = id.index0();
650            idx < len && self.nodes[idx].stamp == id.stamp() && !self.nodes[idx].is_removed()
651        };
652
653        for (i, node) in self.nodes.iter().enumerate() {
654            if node.is_removed() {
655                continue;
656            }
657
658            if !matches!(node.data, NodeData::Data(_)) {
659                return false;
660            }
661
662            if let Some(parent) = node.parent {
663                if !is_valid(parent) {
664                    return false;
665                }
666            }
667
668            if let Some(prev) = node.previous_sibling {
669                if !is_valid(prev)
670                    || self.nodes[prev.index0()].next_sibling.map(|n| n.index0()) != Some(i)
671                {
672                    return false;
673                }
674            }
675            if let Some(next) = node.next_sibling {
676                if !is_valid(next) {
677                    return false;
678                }
679                let next_node = &self.nodes[next.index0()];
680                if next_node.previous_sibling.map(|n| n.index0()) != Some(i)
681                    || next_node.parent.map(|n| n.index0()) != node.parent.map(|n| n.index0())
682                {
683                    return false;
684                }
685            }
686
687            if node.first_child.is_some() != node.last_child.is_some() {
688                return false;
689            }
690
691            if let Some(first) = node.first_child {
692                if !is_valid(first) {
693                    return false;
694                }
695                let mut child = Some(first);
696                let mut last_seen = first;
697                let mut steps = 0;
698                while let Some(c) = child {
699                    let idx = c.index0();
700                    if idx >= len {
701                        return false;
702                    }
703                    let child_node = &self.nodes[idx];
704                    if child_node.parent.map(|n| n.index0()) != Some(i) {
705                        return false;
706                    }
707                    last_seen = c;
708                    child = child_node.next_sibling;
709                    steps += 1;
710                    if steps > len {
711                        return false;
712                    }
713                }
714                if node.last_child != Some(last_seen) {
715                    return false;
716                }
717            }
718        }
719
720        // Every live node must be reachable from the first node of a root
721        // sibling chain by following `first_child` and `next_sibling` links,
722        // and no node may be reached twice. Otherwise a node is part of a
723        // cycle (e.g. two nodes being each other's parent) or is not linked
724        // from its parent's child chain.
725        let mut visited = vec![false; len];
726        let mut stack: Vec<usize> = self
727            .nodes
728            .iter()
729            .enumerate()
730            .filter(|(_, node)| {
731                !node.is_removed() && node.parent.is_none() && node.previous_sibling.is_none()
732            })
733            .map(|(i, _)| i)
734            .collect();
735        let mut reached = 0;
736        while let Some(i) = stack.pop() {
737            if mem::replace(&mut visited[i], true) {
738                return false;
739            }
740            reached += 1;
741            let node = &self.nodes[i];
742            stack.extend(node.next_sibling.map(NodeId::index0));
743            stack.extend(node.first_child.map(NodeId::index0));
744        }
745        // Count the live nodes directly instead of trusting any bookkeeping,
746        // since this is used to check arenas that might be inconsistent.
747        if reached != self.nodes.iter().filter(|n| !n.is_removed()).count() {
748            return false;
749        }
750
751        // Validate free list
752        if self.first_free_slot.is_some() != self.last_free_slot.is_some() {
753            return false;
754        }
755        let mut free_count = 0;
756        let mut last_visited = None;
757        let mut slot = self.first_free_slot;
758        while let Some(idx) = slot {
759            if idx >= len {
760                return false;
761            }
762            let node = &self.nodes[idx];
763            if !node.is_removed() {
764                return false;
765            }
766            match node.data {
767                NodeData::NextFree(next) => slot = next,
768                _ => return false,
769            }
770            last_visited = Some(idx);
771            free_count += 1;
772            if free_count > len {
773                return false;
774            }
775        }
776
777        self.last_free_slot == last_visited
778    }
779
780    pub(crate) fn free_node(&mut self, id: NodeId) {
781        let node = &mut self[id];
782        if node.is_removed() {
783            return;
784        }
785        node.data = NodeData::NextFree(None);
786        node.stamp.mark_removed();
787        node.parent = None;
788        node.previous_sibling = None;
789        node.next_sibling = None;
790        node.first_child = None;
791        node.last_child = None;
792        let stamp = node.stamp;
793        if stamp.reuseable() {
794            if let Some(index) = self.last_free_slot {
795                let new_last = id.index0();
796                self.nodes[index].data = NodeData::NextFree(Some(new_last));
797                self.last_free_slot = Some(new_last);
798            } else {
799                debug_assert!(self.first_free_slot.is_none());
800                debug_assert!(self.last_free_slot.is_none());
801                self.first_free_slot = Some(id.index0());
802                self.last_free_slot = Some(id.index0());
803            }
804        }
805    }
806
807    fn pop_front_free_node(&mut self) -> Option<usize> {
808        let first = self.first_free_slot.take();
809        if let Some(index) = first {
810            if let NodeData::NextFree(next_free) = self.nodes[index].data {
811                self.first_free_slot = next_free;
812            } else {
813                unreachable!("A data node considered as a freed node");
814            }
815            if self.first_free_slot.is_none() {
816                self.last_free_slot = None;
817            }
818        }
819
820        first
821    }
822}

827    ///
828    /// Requires the `par_iter` feature. Uses [rayon](https://docs.rs/rayon)

841    /// root.append_value(2, &mut arena);
842    /// root.append_value(3, &mut arena);
843    ///
844    /// let sum: i64 = arena.par_iter().map(|node| *node.get()).sum();
845    /// assert_eq!(sum, 6);
846    /// ```
847    ///
848    /// [`is_removed()`]: Node::is_removed
849    pub fn par_iter(&self) -> rayon::slice::Iter<'_, Node<T>> {
850        self.nodes.par_iter()
851    }

947/// Index by [`NodeId`] for convenient `arena[id]` access.
948///
949/// Unlike [`Arena::get`], this does **not** validate the node's stamp,
950/// so it may silently return data from a reused slot if the `NodeId`
951/// is stale. For safe access, prefer [`Arena::get`] or [`Arena::get_mut`].
952///
953/// # Panics
954///
955/// Panics if `node` is out of bounds. Note that indexing does not validate
956/// that the `NodeId` originated from this arena. Using an ID from a
957/// different arena may silently access the wrong node or panic.
958impl<T> Index<NodeId> for Arena<T> {
959    type Output = Node<T>;
960
961    #[inline]
962    fn index(&self, node: NodeId) -> &Node<T> {
963        &self.nodes[node.index0()]
964    }
965}

967/// Mutable index by [`NodeId`].
968///

970/// For safe access, prefer [`Arena::get_mut`].
971///
972/// # Panics
973///
974/// Panics if `node` is out of bounds.
975impl<T> IndexMut<NodeId> for Arena<T> {
976    #[inline]
977    fn index_mut(&mut self, node: NodeId) -> &mut Node<T> {
978        &mut self.nodes[node.index0()]
979    }
